cmake_minimum_required(VERSION 3.13)

project(AtomVM)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../CMakeModules")

# Require a cross-compilation toolchain
if (NOT CMAKE_TOOLCHAIN_FILE)
    message(FATAL_ERROR
        "Cross compiling only. Please use:\n"
        "  cmake -S src/platforms/wasi -B build-wasi \\\n"
        "    -DCMAKE_TOOLCHAIN_FILE=cmake/wasi-sdk.cmake\n"
        "Or set WASI_SDK_PATH environment variable first."
    )
endif()

# Options that make sense for this platform
option(AVM_VERBOSE_ABORT "Print module and line number on VM abort" OFF)
option(AVM_CREATE_STACKTRACES "Create stacktraces" ON)
option(AVM_USE_32BIT_FLOAT "Use 32 bit floats." OFF)

# WASI constraints: no SMP, no JIT, no task driver
set(AVM_DISABLE_SMP ON FORCE)
set(AVM_DISABLE_TASK_DRIVER ON FORCE)
set(AVM_DISABLE_JIT ON FORCE)

# WASM requires optimization to keep the local variable count under the
# 50 000 limit mandated by the spec.  The giant switch in
# opcodesswitch.h (scheduler_entry_point) exceeds it at -O0.
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Oz" CACHE STRING "" FORCE)

# WASI does not support these POSIX features
set(HAVE_MKFIFO "" CACHE INTERNAL "Have symbol mkfifo" FORCE)
set(HAVE_UNLINK "" CACHE INTERNAL "Have symbol unlink" FORCE)
set(HAVE_EXECVE "" CACHE INTERNAL "Have symbol execve" FORCE)
set(HAVE_CLOSEFROM "" CACHE INTERNAL "Have symbol closefrom" FORCE)
# WASI p2 provides sockets and select via wasi:sockets
# HAVE_SOCKET and HAVE_SELECT are auto-detected by libAtomVM

set(
    PLATFORM_LIB_SUFFIX
    ${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}
)

# Build the AtomVM executable
add_executable(AtomVM main.c)
set_target_properties(AtomVM PROPERTIES
    OUTPUT_NAME "AtomVM"
    SUFFIX ".wasm"
)
target_compile_features(AtomVM PUBLIC c_std_11)
target_compile_options(AtomVM PUBLIC -Wall -Wextra -Oz)

# Build libAtomVM (the core runtime)
add_subdirectory(../../libAtomVM libAtomVM)

# WASI libc excludes tzset() (no timezone database). libAtomVM's nifs.c
# calls tzset() for erlang:localtime/0 — provide it via a compat header
# that force-includes a static inline no-op before any source file.
target_compile_options(libAtomVM PRIVATE
    -include "${CMAKE_CURRENT_SOURCE_DIR}/lib/wasi_compat.h"
)

target_link_libraries(AtomVM PUBLIC libAtomVM)

# Build the WASI platform library
add_subdirectory(lib)
target_link_libraries(AtomVM PRIVATE libAtomVM${PLATFORM_LIB_SUFFIX})
target_include_directories(AtomVM PRIVATE lib/)

# ---------------------------------------------------------------------------
# AtomVM HTTP Component (wasi:http/incoming-handler for Spin)
# ---------------------------------------------------------------------------
option(AVM_BUILD_HTTP_COMPONENT "Build the AtomVM HTTP component for Spin" ON)

if(AVM_BUILD_HTTP_COMPONENT)
    set(HTTP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/http")

    add_executable(AtomVM_http
        ${HTTP_DIR}/main_http.c
        ${HTTP_DIR}/wasi_http_handler.c
        ${HTTP_DIR}/generated/app.c
    )
    set_target_properties(AtomVM_http PROPERTIES
        OUTPUT_NAME "AtomVM_http"
        SUFFIX ".wasm"
    )
    target_compile_features(AtomVM_http PUBLIC c_std_11)
    target_compile_options(AtomVM_http PUBLIC -Wall -Wextra -Oz)
    target_include_directories(AtomVM_http PRIVATE
        ${HTTP_DIR}
        ${HTTP_DIR}/generated
        lib/
    )
    target_link_libraries(AtomVM_http PUBLIC libAtomVM)
    target_link_libraries(AtomVM_http PRIVATE libAtomVM${PLATFORM_LIB_SUFFIX})

    # Link the component type object (generated by wit-bindgen)
    target_link_libraries(AtomVM_http PRIVATE
        ${HTTP_DIR}/generated/app_component_type.o
    )

    # Reactor mode: no _start entry point, use constructors for init
    target_link_options(AtomVM_http PRIVATE
        -mexec-model=reactor
        -Wl,--export=cabi_realloc
    )

    # Force-include WASI compat shims
    target_compile_options(AtomVM_http PRIVATE
        -include "${CMAKE_CURRENT_SOURCE_DIR}/lib/wasi_compat.h"
    )
endif()
